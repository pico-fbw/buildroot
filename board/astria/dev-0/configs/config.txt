# For more options and information see http://rptl.io/configtxt

start_file=start4.elf
fixup_file=fixup4.dat
kernel=Image

# 64 bit support
arm_64bit=1

# Enable serial console on boot
enable_uart=1
uart_2ndstage=1
dtdebug=1

# Enable DWC USB controller (USB 2.0) in device mode
dtoverlay=dwc2,dr_mode=device

# Disable unused interfaces, enable additional interfaces
dtparam=hdmi=off,rtc=off
dtparam=i2c_arm=on,i2c_arm_baudrate=400000,spi=on
dtoverlay=i2c4,pins_6_7=1,baudrate=400000
dtoverlay=spi1-2cs
dtoverlay=uart4
dtoverlay=uart5

# Enable hardware drivers
dtoverlay=ads1115
dtoverlay=i2c-pwm-pca9685a
dtoverlay=i2c_rtc,mcp7941x
dtoverlay=mcp251xfd,spi1-0,interrupt=25

# Automatically load overlays for detected cameras
camera_auto_detect=1

# Automatically load initramfs files, if found
auto_initramfs=1

# Run as fast as firmware / board allows
arm_boost=1
