# For more options and information see http://rptl.io/configtxt

start_file=start4.elf
fixup_file=fixup4.dat
kernel=Image

# 64 bit support
arm_64bit=1

# Enable serial console on boot
enable_uart=1
uart_2ndstage=1

# Enable DWC USB controller (USB 2.0) in device mode
dtoverlay=dwc2,dr_mode=device

# Enable additional hardware interfaces
dtparam=i2c_arm=on,i2c_arm_baudrate=400000,spi=on
dtoverlay=i2c-gpio,i2c_gpio_sda=6,i2c_gpio_scl=7
dtoverlay=spi0-2cs
dtoverlay=uart5

# Automatically load overlays for detected cameras
camera_auto_detect=1

# Automatically load initramfs files, if found
auto_initramfs=1

# Enable DRM VC4 V3D driver
dtoverlay=vc4-kms-v3d
max_framebuffers=2

# Don't have the firmware create an initial video= setting in cmdline.txt,
# use the kernel's default instead
disable_fw_kms_setup=1

# Disable compensation for displays with overscan
disable_overscan=1

# Run as fast as firmware / board allows
arm_boost=1
